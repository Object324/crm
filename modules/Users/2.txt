 SELECT  distinct ld_account.accountname,ld_account.smownerid,ld_account.account_no,ld_account.accountid,ld_account.publicaccount FROM ld_account left join    ld_users as  users_assigned_user_id  on  users_assigned_user_id .id=ld_account.smownerid inner join ( SELECT distinct
            		ld_account.accountid
            	FROM
            		ld_account  left join   ld_users as  users_assigned_user_id  on  users_assigned_user_id .id=ld_account.smownerid WHERE ld_account.deleted = 0  AND   (  (( ld_account.accountname LIKE  '%成都瀚德科技%')  or ( ifnull( users_assigned_user_id .last_name,'') LIKE  '%成都瀚德科技%')  or ( ld_account.account_no LIKE  '%成都瀚德科技%') )) AND ld_account.accountid > 0 and( (ld_account.publicaccount=0  and (ld_account.smownerid in (49,338414,49) or ld_account.smcreatorid = 49 or  ld_account.accountid in (SELECT crmid from ld_tabshare where viewuserid = 49) )) or  ( ld_account.pool_id = 1 and ld_account.publicaccount=1  and (ld_account.smownerid in (49,338414,49) or ld_account.smcreatorid = 49 or  ld_account.accountid in (SELECT crmid from ld_tabshare where viewuserid = 49) )) or  ( ld_account.pool_id = 1 and ld_account.publicaccount > 1  and (ld_account.smownerid in (49,338414,49) or ld_account.smcreatorid = 49 or  ld_account.accountid in (SELECT crmid from ld_tabshare where viewuserid = 49) ))) order by FIELD(ld_account.accountname,"成都瀚德科技") DESC, ld_account.createdtime DESC limit 0,5 ) as tmp on tmp.accountid=ld_account.accountid order by FIELD(ld_account.accountname,"成都瀚德科技") DESC, ld_account.createdtime DESC
